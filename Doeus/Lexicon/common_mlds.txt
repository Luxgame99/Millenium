sub = Substantiv
ver = Verb (intransitiv)
adj = Adjektiv
adv = Adverb
pk = Konjunktion bzw. Partikel

sub m = Nominalklasse physisch
sub s = Nominalklasse abstrakt
sub f = Nominalklasse formlos

ver t = transitives Verb
ver m = modales Verb
ver r = reflexives Verb

adj a = neutrales Adjektiv
adj e = positives Adjektiv
adj o = negatives Adjektiv
